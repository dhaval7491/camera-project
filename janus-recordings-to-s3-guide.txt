# Create the conversion script

#!/bin/bash
#===============================================================================
# Script: convert_mjr_to_mp4.sh
# Purpose: Convert Janus MJR recordings to MP4 and upload to S3
#===============================================================================

RECORDINGS_DIR="/opt/janus/share/janus/recordings"
TEMP_DIR="/opt/janus/processing/temp"
CONVERTED_DIR="/opt/janus/processing/converted"
LOG_FILE="/opt/janus/processing/logs/conversion.log"
S3_BUCKET="your-camera-recordings-bucket"
LARAVEL_API_URL="http://your-laravel-app.com/api/recording/store"
API_TOKEN="your-api-token"

# Function to log messages
log_message() {
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] $1" >> "$LOG_FILE"
}

# Function to extract camera ID and timestamp from filename
parse_filename() {
    local filename=$1
    # Format: 23_Publisher_2025-09-03T19-45-54-audio-0.mjr
    local camera_id=$(echo "$filename" | cut -d'_' -f1)
    local timestamp=$(echo "$filename" | cut -d'_' -f3 | cut -d'-' -f1-6)
    echo "$camera_id|$timestamp"
}

# Function to convert MJR to MP4
convert_recording() {
    local camera_id=$1
    local timestamp=$2

    log_message "Starting conversion for Camera ID: $camera_id, Timestamp: $timestamp"

    # Find audio and video files
    local audio_file="${RECORDINGS_DIR}/${camera_id}_Publisher_${timestamp}-audio-0.mjr"
    local video_file="${RECORDINGS_DIR}/${camera_id}_Publisher_${timestamp}-video-1.mjr"

    if [[ ! -f "$audio_file" ]] || [[ ! -f "$video_file" ]]; then
        log_message "ERROR: Missing audio or video file for $camera_id at $timestamp"
        return 1
    fi

    # Convert MJR to raw formats
    local temp_audio="${TEMP_DIR}/${camera_id}_${timestamp}.opus"
    local temp_video="${TEMP_DIR}/${camera_id}_${timestamp}.webm"

    janus-pp-rec "$audio_file" "$temp_audio"
    janus-pp-rec "$video_file" "$temp_video"

    # Merge audio and video into MP4
    local output_file="${CONVERTED_DIR}/${camera_id}_${timestamp}.mp4"

    ffmpeg -i "$temp_video" -i "$temp_audio" \
           -c:v libx264 -crf 23 -preset medium \
           -c:a aac -b:a 128k \
           -movflags +faststart \
           "$output_file" -y

    if [[ $? -eq 0 ]]; then
        log_message "Conversion successful: $output_file"

        # Upload to S3
        upload_to_s3 "$output_file" "$camera_id" "$timestamp"

        # Clean up temporary files
        rm -f "$temp_audio" "$temp_video"
        rm -f "$audio_file" "$video_file"

        return 0
    else
        log_message "ERROR: Conversion failed for $camera_id at $timestamp"
        return 1
    fi
}

# Function to upload to S3
upload_to_s3() {
    local file_path=$1
    local camera_id=$2
    local timestamp=$3
    local filename=$(basename "$file_path")

    # Create S3 path: camera_id/year/month/filename
    local year=$(echo "$timestamp" | cut -d'-' -f1)
    local month=$(echo "$timestamp" | cut -d'-' -f2)
    local s3_path="recordings/${camera_id}/${year}/${month}/${filename}"

    log_message "Uploading to S3: s3://${S3_BUCKET}/${s3_path}"

    aws s3 cp "$file_path" "s3://${S3_BUCKET}/${s3_path}" \
        --metadata "camera_id=${camera_id},timestamp=${timestamp}"

    if [[ $? -eq 0 ]]; then
        log_message "Upload successful: ${s3_path}"

        # Update Laravel database
        update_database "$camera_id" "$timestamp" "$s3_path"

        # Remove local converted file
        rm -f "$file_path"

        return 0
    else
        log_message "ERROR: Upload failed for $filename"
        return 1
    fi
}

# Function to update Laravel database
update_database() {
    local camera_id=$1
    local timestamp=$2
    local s3_path=$3

    # Convert timestamp format for database
    local db_timestamp=$(echo "$timestamp" | sed 's/T/ /' | tr '-' ':' | cut -d' ' -f1-2)

    # Make API call to Laravel
    curl -X POST "${LARAVEL_API_URL}" \
         -H "Authorization: Bearer ${API_TOKEN}" \
         -H "Content-Type: application/json" \
         -d "{
             \"camera_id\": \"${camera_id}\",
             \"recording_timestamp\": \"${db_timestamp}\",
             \"recording_name\": \"${s3_path}\"
         }"

    if [[ $? -eq 0 ]]; then
        log_message "Database updated for recording: ${s3_path}"
    else
        log_message "ERROR: Database update failed for ${s3_path}"
    fi
}

# Main processing function
process_recordings() {
    # Find all audio MJR files (use audio as reference)
    for audio_file in ${RECORDINGS_DIR}/*-audio-*.mjr; do
        if [[ -f "$audio_file" ]]; then
            filename=$(basename "$audio_file")
            parsed=$(parse_filename "$filename")
            camera_id=$(echo "$parsed" | cut -d'|' -f1)
            timestamp=$(echo "$parsed" | cut -d'|' -f2)

            convert_recording "$camera_id" "$timestamp"
        fi
    done
}

# Check if script is already running
if [[ -f "/var/run/mjr_converter.pid" ]]; then
    OLD_PID=$(cat /var/run/mjr_converter.pid)
    if ps -p $OLD_PID > /dev/null 2>&1; then
        log_message "Converter already running with PID $OLD_PID"
        exit 0
    fi
fi

# Save PID
echo $$ > /var/run/mjr_converter.pid

# Run the processor
log_message "Starting MJR to MP4 converter"
process_recordings
log_message "Conversion process completed"

# Remove PID file
rm -f /var/run/mjr_converter.pid

# Create watcher script for real-time processing

#!/bin/bash
#===============================================================================
# Script: watch_recordings.sh
# Purpose: Monitor recordings directory and trigger conversion
#===============================================================================

RECORDINGS_DIR="/opt/janus/share/janus/recordings"
CONVERTER_SCRIPT="/opt/janus/processing/scripts/convert_mjr_to_mp4.sh"
LOG_FILE="/opt/janus/processing/logs/watcher.log"

log_message() {
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] $1" >> "$LOG_FILE"
}

log_message "Starting recordings watcher"

# Monitor for new .mjr files
inotifywait -m -r -e close_write --format '%w%f' "$RECORDINGS_DIR" | while read FILE
do
    if [[ "$FILE" == *"-audio-0.mjr" ]]; then
        log_message "New recording detected: $FILE"

        # Wait for video file to complete
        sleep 5

        # Trigger conversion
        $CONVERTER_SCRIPT &
    fi
done

# Make script executable
sudo chmod +x /opt/janus/processing/scripts/watch_recordings.sh

<?php
namespace App\Http\Controllers\Api;

use App\Http\Controllers\Controller;
use App\Models\Recording;
use Illuminate\Http\Request;

class RecordingController extends Controller
{
    public function store(Request $request)
    {
        $validated = $request->validate([
            'camera_id' => 'required|exists:equipments,id',
            'recording_timestamp' => 'required|string',
            'recording_name' => 'required|string'
        ]);

        $recording = Recording::create($validated);

        return response()->json([
            'success' => true,
            'recording' => $recording
        ], 201);
    }
}

# Update Recording Model
<?php

use Illuminate\Database\Eloquent\Model;

class Recording extends Model
{

        'camera_id',
        'recording_timestamp',
        'recording_name'
    ];

    public function camera()
    {
        return $this->belongsTo(Equipment::class, 'camera_id');
    }
}

====================================================================================================
TROUBLESHOOTING
====================================================================================================

1. Permission Issues:
   - Ensure www-data user has access to all directories
   - Check SELinux/AppArmor policies

2. Conversion Failures:
   - Check janus-pp-rec is installed: which janus-pp-rec
   - Verify FFmpeg codecs: ffmpeg -codecs

3. S3 Upload Issues:
   - Verify AWS credentials: aws s3 ls
   - Check bucket permissions and policies

4. Database Connection:
   - Verify Laravel API endpoint is accessible
   - Check API token is valid
   - Review Laravel logs: tail -f storage/logs/laravel.log

5. Service Not Starting:
   - Check systemd logs: journalctl -u janus-recording-processor -f
   - Verify script paths are correct
   - Check file permissions
